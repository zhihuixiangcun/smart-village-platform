### 任务3: 实现二维码缓存优化

#### 需求
避免重复调用API生成相同的二维码，提升性能。

#### 实现方案

**1. LocalStorage缓存**
```javascript
// composables/useQRCodeCache.js
const QR_CACHE_KEY = 'qrcode_cache'
const QR_CACHE_DURATION = 24 * 60 * 60 * 1000 // 24小时

export function useQRCodeCache() {
  const getCachedQRCode = (householdId) => {
    const cache = localStorage.getItem(QR_CACHE_KEY)
    if (!cache) return null

    const parsed = JSON.parse(cache)
    const item = parsed[householdId]

    if (!item) return null

    // 检查是否过期
    if (Date.now() - item.timestamp > QR_CACHE_DURATION) {
      delete parsed[householdId]
      localStorage.setItem(QR_CACHE_KEY, JSON.stringify(parsed))
      return null
    }

    return item.data
  }

  const setCachedQRCode = (householdId, data) => {
    const cache = localStorage.getItem(QR_CACHE_KEY)
    const parsed = cache ? JSON.parse(cache) : {}

    parsed[householdId] = {
      data,
      timestamp: Date.now()
    }

    localStorage.setItem(QR_CACHE_KEY, JSON.stringify(parsed))
  }

  return { getCachedQRCode, setCachedQRCode }
}
```

**2. VillageAffairsView集成**
```javascript
import { useQRCodeCache } from '@/composables/useQRCodeCache'

const { getCachedQRCode, setCachedQRCode } = useQRCodeCache()

const showQRCode = async () => {
  qrcodeDialogVisible.value = true
  qrCodeLoading.value = true

  try {
    const householdId = userStore.userInfo?.householdId

    // 先尝试从缓存获取
    const cached = getCachedQRCode(householdId)
    if (cached) {
      userQRCode.value = cached.qrImageUrl
      householdInfo.value = cached.household
      qrCodeLoading.value = false
      ElMessage.success('从缓存加载二维码')
      return
    }

    // 缓存未命中，调用API
    const response = await householdQRApi.generateQR(householdId, { includeImage: true })

    if (response.success) {
      userQRCode.value = response.data.qrImageUrl
      householdInfo.value = response.data.household

      // 保存到缓存
      setCachedQRCode(householdId, response.data)

      ElMessage.success('二维码生成成功')
    }
  } catch (error) {
    ElMessage.error('生成二维码失败，请稍后重试')
  } finally {
    qrCodeLoading.value = false
  }
}
```

const refreshQRCode = async (force = false) => {
  const householdId = userStore.userInfo?.householdId

  if (!force) {

### 任务4: 实现人脸识别验证功能

#### 需求
扫码后需要人脸识别验证才能查看完整的敏感信息。

#### 实现方案

**1. 后端API**
```javascript
// controllers/faceRecognitionController.js
const faceRecognitionService = require('../services/faceRecognitionService')

class FaceRecognitionController {
  /**
   * 人脸识别验证
   * POST /api/v1/face-recognition/verify
   */
  verifyFace = async (req, res) => {
    try {
      const { faceImageBase64, householdId } = req.body

      // 调用人脸识别服务
      const result = await faceRecognitionService.verifyFace(
        faceImageBase64,
        householdId
      )

      res.json({
        success: result.verified,
        data: {
          verified: result.verified,
          confidence: result.confidence,
          matchResult: result.matchResult
        }
      })
    } catch (error) {
      res.status(500).json({
        success: false,
        message: error.message
      })
    }
  }
}

module.exports = new FaceRecognitionController()
```

```vue
<!-- components/FaceVerificationDialog.vue -->
<template>

        <el-step title="验证完成" />

      <!-- 步骤1: 拍摄人脸 -->
      <div v-if="currentStep === 0" class="step-content">
        <video
          ref="videoRef"
          autoplay
          :width="400"
          :height="300"
        ></video>
        <canvas
          ref="canvasRef"
          style="display:none"
          :width="400"
          :height="300"

          </el-button>

        </div>

      <!-- 步骤2: 识别验证 -->
      <div v-if="currentStep === 1" class="step-content">
        <div class="verifying">
          <el-icon class="is-loading" :size="50"><Loading /></el-icon>
          <p>正在识别验证...</p>
        </div>
      </div>

      <!-- 步骤3: 验证结果 -->
      <div v-if="currentStep === 2" class="step-content">
        <el-result
          :icon="verifyResult.success ? 'success' : 'error'"
          :title="verifyResult.success ? '验证成功' : '验证失败'"
          :sub-title="verifyResult.message"
        >
          <template #extra>
            <el-button type="primary" @click="handleConfirm">
              确定
            </el-button>
            <el-button v-if="!verifyResult.success" @click="retry">
              重试
            </el-button>
          </template>
        </el-result>
      </div>
    </div>
  </el-dialog>
</template>

<script setup>
import { ref } from 'vue'
import { ElMessage } from 'element-plus'
import axios from 'axios'

const props = defineProps({
  householdId: String
})

const emit = defineEmits(['verified', 'cancel'])

const visible = ref(true)
const currentStep = ref(0)
const capturing = ref(false)
const videoRef = ref(null)
const canvasRef = ref(null)
const verifyResult = ref({})

// 启动摄像头
const startCamera = async () => {
  try {
    const stream = await navigator.mediaDevices.getUserMedia({ video: true })
    videoRef.value.srcObject = stream
  } catch (error) {
    ElMessage.error('无法访问摄像头')
    visible.value = false
  }
}

// 拍摄人脸
const captureFace = () => {
  capturing.value = true
  const canvas = canvasRef.value
  const video = videoRef.value

  canvas.getContext('2d').drawImage(video, 0, 0)
  const faceImageBase64 = canvas.toDataURL('image/jpeg')

  // 停止摄像头
  video.srcObject.getTracks().forEach(track => track.stop())

  // 进入验证步骤
  currentStep.value = 1
  verifyFace(faceImageBase64)
}

// 验证人脸
const verifyFace = async (faceImageBase64) => {
  try {
    const response = await axios.post('/api/v1/face-recognition/verify', {
      faceImageBase64,
      householdId: props.householdId
    })

    verifyResult.value = response.data
    currentStep.value = 2

    if (response.data.success) {
      emit('verified', response.data.data)
    }
  } catch (error) {
    verifyResult.value = {
      success: false,
      message: '验证失败，请重试'
    }
    currentStep.value = 2
  }
}

// 重试
const retry = () => {
  currentStep.value = 0
  verifyResult.value = {}
  startCamera()
}

// 确认
const handleConfirm = () => {
  visible.value = false
  if (verifyResult.value.success) {
    emit('verified', verifyResult.value)
  } else {
    emit('cancel')
  }
}

// 初始化
startCamera()
</script>
```

    return

  progress.value = 0
  results.value = []

    if (response.success) {
      const { results: batchResults, summary } = response.data

      totalCount.value = summary.total
      successCount.value = summary.success
      failCount.value = summary.failed

      // 模拟进度动画
      const animateProgress = () => {
        const interval = setInterval(() => {
          if (progress.value < 100) {
            progress.value += 10
          } else {
            clearInterval(interval)
            // 显示结果
            results.value = batchResults
            completedCount.value = totalCount.value
          }
        }, 200)
      }

      animateProgress()

      ElMessage.success(`批量生成完成！成功: ${successCount.value}, 失败: ${failCount.value}`)
    }
  } catch (error) {
    progressStatus.value = 'exception'
    ElMessage.error('批量生成失败')
  } finally {
    generating.value = false
  }
}

// 导出结果
const handleExport = () => {

    r.codeId,
    r.status === 'success' ? '成功' : '失败',
    r.message

  rows.forEach(row => {
    csvContent += row.join(',') + '\n'

  // 下载

  const link = document.createElement('a')
  link.href = URL.createObjectURL(blob)
  link.download = `批量生成户码_${new Date().toISOString()}.csv`

### 任务7: 实现离线功能

#### 需求
在无网络环境下也能显示和使用二维码。

#### 实现方案

**1. Service Worker注册**
```javascript
// public/sw.js
const CACHE_NAME = 'smart-village-v1'
const QR_CACHE_PREFIX = 'qr-code-'

// 安装Service Worker
self.addEventListener('install', (event) => {
  event.waitUntil(
    caches.open(CACHE_NAME).then((cache) => {
      return cache.addAll([
        '/',
        '/offline.html',
        '/manifest.json'
      ])
    })
  )
})

// 拦截请求
self.addEventListener('fetch', (event) => {
  event.respondWith(
    caches.match(event.request).then((response) => {
      return response || fetch(event.request)
    })
  )
})

// 监听消息
self.addEventListener('message', (event) => {
  if (event.data.type === 'CACHE_QR_CODE') {
    cacheQRCode(event.data.householdId, event.data.qrData)
  }
})

// 缓存二维码
function cacheQRCode(householdId, qrData) {
  caches.open(CACHE_NAME).then((cache) => {
    const url = `${QR_CACHE_PREFIX}${householdId}`
    const response = new Response(JSON.stringify(qrData), {
      headers: { 'Content-Type': 'application/json' }
    })
    cache.put(url, response)
  })
}
```

**2. 前端集成**
```javascript
// composables/useOfflineQR.js
export function useOfflineQR() {
  const isOnline = ref(navigator.onLine)

  // 监听网络状态
  window.addEventListener('online', () => {
    isOnline.value = true
    ElMessage.success('网络已连接')
  })

  window.addEventListener('offline', () => {
    isOnline.value = false
    ElMessage.warning('网络已断开，切换到离线模式')
  })

  // 保存二维码到IndexedDB
  const saveQRToIndexedDB = async (householdId, qrData) => {
    if (!window.indexedDB) return false

    return new Promise((resolve, reject) => {
      const request = indexedDB.open('SmartVillageDB', 1)

      request.onerror = () => reject('数据库打开失败')
      request.onsuccess = () => {
        const db = request.result
        const transaction = db.transaction(['qrcodes'], 'readwrite')
        const store = transaction.objectStore('qrcodes')

        const item = {
          householdId,
          qrData,
          timestamp: Date.now()
        }

        const putRequest = store.put(item)

        putRequest.onsuccess = () => resolve(true)
        putRequest.onerror = () => reject('保存失败')
      }
    })
  }

  // 从IndexedDB获取二维码
  const getQRFromIndexedDB = async (householdId) => {
    if (!window.indexedDB) return null

    return new Promise((resolve) => {
      const request = indexedDB.open('SmartVillageDB', 1)

      request.onsuccess = () => {
        const db = request.result
        const transaction = db.transaction(['qrcodes'], 'readonly')
        const store = transaction.objectStore('qrcodes')
        const getRequest = store.get(householdId)

        getRequest.onsuccess = () => {
          const result = getRequest.result
          if (result) {
            // 检查是否过期（7天）
            const isExpired = Date.now() - result.timestamp > 7 * 24 * 60 * 60 * 1000
            if (isExpired) {
              resolve(null)
            } else {
              resolve(result.qrData)
            }
          } else {
            resolve(null)
          }
        }
      }
    })
  }

  // 生成并缓存二维码
  const generateAndCacheQR = async (householdId) => {
    try {
      // 尝试在线生成
      if (isOnline.value) {
        const response = await householdQRApi.generateQR(householdId, { includeImage: true })

        if (response.success) {
          // 保存到IndexedDB
          await saveQRToIndexedDB(householdId, response.data)

          // 通知Service Worker
          if (navigator.serviceWorker.controller) {
            navigator.serviceWorker.controller.postMessage({
              type: 'CACHE_QR_CODE',
              householdId,
              qrData: response.data
            })
          }

          return response.data
        }
      } else {
        // 离线模式，从IndexedDB获取
        const cached = await getQRFromIndexedDB(householdId)

        if (cached) {
          ElMessage.info('使用离线缓存的二维码')
          return cached
        } else {
          throw new Error('无离线缓存且网络不可用')
        }
      }
    } catch (error) {
      ElMessage.error(error.message)
      throw error
    }
  }

  return {
    isOnline,
    saveQRToIndexedDB,
    getQRFromIndexedDB,
    generateAndCacheQR
  }
}
```
